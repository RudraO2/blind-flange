# Runtime dependencies of the ingestion service itself (services/ingestion/server.py,
# ocr.py) — distinct from ../requirements-fixtures.txt (the sample-report generator,
# build-time only) and proof/requirements-proof.txt (the timeboxed engine proofs).
#
# The HTTP boundary is stdlib `http.server` deliberately — no web framework.
#
# Licences verified from the LICENSE file at these exact versions and recorded in
# LICENCES.md, per docs/licence-policy.md.

# The OCR engine, replacing Tesseract on 28 Aug 2026 — see proof/PROOF-RAPIDOCR.md for the
# measurements and for the one open licence question this introduces (pyclipper embeds the
# Clipper C++ library under the Boost Software License, which is not on the allow-list).
# Ships its three PP-OCRv6 ONNX models inside the wheel; proof/rapidocr_proof.py verifies a
# full pass makes no outbound connection and no DNS lookup.
#
# Install with --no-deps if you would rather not have pip fetch shapely at all; the six
# packages below are the complete set rapidocr actually needs.
rapidocr==3.9.2

# The inference engine rapidocr runs those models on. CPU build — no CUDA, no VRAM.
onnxruntime==1.24.4

# Pulled in by rapidocr. Recorded here rather than left transitive, because docs/licence-policy.md
# applies to the whole dependency tree and an unpinned transitive is an unverified licence.
#
# `shapely` is deliberately NOT pinned here and must be uninstalled if pip pulls it in:
#
#     pip install -r requirements.txt && pip uninstall -y shapely
#
# The shapely wheel bundles the GEOS shared libraries, and GEOS is LGPL-2.1 — weak copyleft,
# outside the allow-list. RapidOCR's detector uses shapely for exactly one thing, the area
# and perimeter of a detected text box, so ocr.py supplies those two properties itself and
# registers them under `shapely` before RapidOCR can import the real one. Verified to give
# identical output; test_service.py::test_geos_is_never_loaded holds the line.
pyclipper==1.4.0
omegaconf==2.3.1
antlr4-python3-runtime==4.9.3
colorlog==6.12.0

# Both were missing from this file until Story 6.4's audit measured what a real OCR pass
# actually loads, and found `cv2` and `numpy` in the process. requirements.txt's own rule
# — "an unpinned transitive is an unverified licence" — was being broken by the two
# heaviest packages in the tree, one of which bundles FFmpeg. Pinned now, licences in
# LICENCES.md, the FFmpeg question decided in docs/licence-decisions.json.
opencv-python==4.13.0.92
numpy==2.4.3

# NOT pinned, and sealed out at import: `requests`, `tqdm`, and the `certifi` / `urllib3` /
# `idna` stack requests brings with it.
#
# `certifi` is MPL-2.0 and `tqdm` is MPL-2.0 AND MIT — weak copyleft, outside the
# allow-list even after ADR-0006 widened it. RapidOCR imports both at module scope, but
# only ever calls them from its model downloader and its load-image-from-URL branch. The
# models ship inside the wheel and this service is handed decoded images, so neither branch
# is reachable. `ocr.py::_seal_out_http` registers raising stubs under those names before
# RapidOCR can import the real ones, exactly as it already does for shapely.
#
# Verified: 97 regions at 99.68 mean confidence with certifi, urllib3 and idna never
# entering sys.modules. `test_service.py::test_http_client_is_never_loaded` holds the line.
# Removing an HTTP client from an air-gapped product is worth doing for NFR2 alone.

# Image decoding. Its licence (MIT-CMU) is the open question recorded in LICENCES.md since
# Story 4.1 — not resolved by this change, not reopened by it either.
pillow==11.3.0

# Renders each PDF page to an image ocr.py can read. Never PyMuPDF: it is AGPL-3.0
# (ADR-0005, CLAUDE.md). Same version already licence-checked for the fixture generator in
# ../requirements-fixtures.txt.
pypdfium2==4.30.0
